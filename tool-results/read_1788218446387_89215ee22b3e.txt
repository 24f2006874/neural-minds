    55→type ScalingPoint = { districts: number; patientsPerYear: number };

    64→function clamp(v: number, lo: number, hi: number) {
    65→  return Math.min(hi, Math.max(lo, v));
    66→}
    67→
    68→/** Trust-color load scale: green ≤75 · amber 75–90 · red >90. */
    69→function loadTone(pct: number): Tone {
    70→  if (pct > 90) return { text: "text-[#F87171] text-glow-red", chip: "border-[#F87171]/40 text-[#F87171]" };
    71→  if (pct > 75) return { text: "text-[#FBBF24]", chip: "border-[#FBBF24]/40 text-[#FBBF24]" };
    72→  return { text: "text-[#34D399]", chip: "border-[#34D399]/40 text-[#34D399]" };
    73→}
    74→
    75→/** Wait scale: green ≤15 min · amber >30 · red >60 (or unbounded). */
    76→function waitTone(min: number): Tone {
    77→  if (min >= UNBOUNDED_WAIT || min > 60)
    78→    return { text: "text-[#F87171] text-glow-red", chip: "border-[#F87171]/40 text-[#F87171]" };
    79→  if (min > 30) return { text: "text-[#FBBF24]", chip: "border-[#FBBF24]/40 text-[#FBBF24]" };
    80→  if (min <= 15) return { text: "text-[#34D399]", chip: "border-[#34D399]/40 text-[#34D399]" };
    81→  return { text: "text-[#22D3EE]", chip: "border-[#22D3EE]/40 text-[#22D3EE]" };
    82→}
    83→
    84→/** Decorative flow dots — travel left → right, speed set by `duration`. */
    85→function FlowDots({
    86→  count,
    87→  duration,
    88→  color,
    89→  reduced,
    90→}: {
    91→  count: number;
    92→  duration: number;
    93→  color: string;
    94→  reduced: boolean;
    95→}) {
    96→  const dots = Array.from({ length: count }, (_, i) => i);
    97→  if (reduced) {
    98→    return (
    99→      <>
   100→        {dots.map((i) => (
   101→          <span
   102→            key={i}
   103→            className="absolute h-1.5 w-1.5 rounded-full"
   104→            style={{
   105→              top: "calc(50% - 3px)",
   106→              left: `${count > 1 ? 8 + (i * 84) / (count - 1) : 46}%`,
   107→              background: color,
   108→              opacity: 0.55,
   109→            }}
   110→          />
   111→        ))}
   112→      </>
   113→    );
   114→  }
   115→  return (
   116→    <>
   117→      {dots.map((i) => (
   118→        <motion.span
   119→          key={i}
   120→          className="absolute h-1.5 w-1.5 rounded-full"
   121→          style={{ top: "calc(50% - 3px)", background: color, boxShadow: `0 0 8px ${color}` }}
   122→          initial={{ left: "-3%" }}
   123→          animate={{ left: "103%" }}
   124→          transition={{ duration, repeat: Infinity, ease: "linear", delay: (i * duration) / count }}
   125→        />
   126→      ))}
   127→    </>
   128→  );
   129→}
