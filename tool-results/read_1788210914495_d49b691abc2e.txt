   112→   112→// Deterministic pseudo-random generator (seeded)
   113→   113→// ────────────────────────────────────────────────────────────
   114→   114→
   115→   115→export function seededRandom(seedStr: string): () => number {
   116→   116→  let h = 2166136261;
   117→   117→  for (let i = 0; i < seedStr.length; i++) {
   118→   118→    h ^= seedStr.charCodeAt(i);
   119→   119→    h = Math.imul(h, 16777619);
   120→   120→  }
   121→   121→  return () => {
   122→   122→    h ^= h << 13;
   123→   123→    h ^= h >>> 17;
   124→   124→    h ^= h << 5;
   125→   125→    return ((h >>> 0) % 100000) / 100000;
   126→   126→  };
   127→   127→}

   129→   129→function clamp(v: number, min: number, max: number) {
   130→   130→  return Math.min(max, Math.max(min, v));
   131→   131→}
   132→   132→
   133→   133→/** Poisson-ish count from a seeded rng */
   134→   134→function poissonCount(rng: () => number, lambda: number): number {
   135→   135→  // Simple approximation good enough for demo counts
   136→   136→  const base = Math.round(lambda * (0.5 + rng()));
   137→   137→  return Math.max(0, base);
   138→   138→}

   140→   140→/** Ring of points around macula for exudates in DME-risk cases */
   141→   141→function ringPoints(cx: number, cy: number, radius: number, count: number, rng: () => number, rBase: number) {
   142→   142→  const pts: Array<{ x: number; y: number; r: number }> = [];
   143→   143→  for (let i = 0; i < count; i++) {
   144→   144→    const a = (i / count) * Math.PI * 2 + rng() * 0.6;
   145→   145→    const rr = radius * (0.75 + rng() * 0.5);
   146→   146→    pts.push({ x: clamp(cx + Math.cos(a) * rr, 0.1, 0.9), y: clamp(cy + Math.sin(a) * rr, 0.1, 0.9), r: rBase * (0.7 + rng() * 0.9) });
   147→   147→  }
   148→   148→  return pts;
   149→   149→}
   150→   150→
   151→   151→function scatterPoints(count: number, rng: () => number, rBase: number, zones: Array<{ cx: number; cy: number; rad: number; w: number }>) {
   152→   152→  const pts: Array<{ x: number; y: number; r: number }> = [];
   153→   153→  for (let i = 0; i < count; i++) {
   154→   154→    // choose a zone weighted
   155→   155→    let z = zones[0];
   156→   156→    const t = rng() * zones.reduce((s, zz) => s + zz.w, 0);
   157→   157→    let acc = 0;
   158→   158→    for (const zz of zones) {
   159→   159→      acc += zz.w;
   160→   160→      if (t <= acc) { z = zz; break; }
   161→   161→    }
   162→   162→    const a = rng() * Math.PI * 2;
   163→   163→    const rr = Math.sqrt(rng()) * z.rad;
   164→   164→    pts.push({
   165→   165→      x: clamp(z.cx + Math.cos(a) * rr, 0.06, 0.94),
   166→   166→      y: clamp(z.cy + Math.sin(a) * rr, 0.06, 0.94),
   167→   167→      r: rBase * (0.6 + rng() * 1.1),
   168→   168→    });
   169→   169→  }
   170→   170→  return pts;
   171→   171→}

   185→   185→export function buildLesions(patientId: string, cfg: LesionConfig): EvidenceResult["lesions"] {
   186→   186→  const rng = seededRandom(`lesions:${patientId}`);
   187→   187→  // Optic disc ~ (0.30, 0.38), macula ~ (0.62, 0.55) in normalized coords
   188→   188→  const disc = { cx: 0.3, cy: 0.38 };
   189→   189→  const macula = { cx: 0.62, cy: 0.55 };
   190→   190→
   191→   191→  const maZones = [
   192→   192→    { cx: macula.cx, cy: macula.cy, rad: 0.26, w: 5 },
   193→   193→    { cx: disc.cx, cy: disc.cy, rad: 0.22, w: 3 },
   194→   194→    { cx: 0.5, cy: 0.5, rad: 0.4, w: 2 },
   195→   195→  ];
   196→   196→  const hemZones = [
   197→   197→    { cx: macula.cx, cy: macula.cy, rad: 0.3, w: 4 },
   198→   198→    { cx: disc.cx, cy: disc.cy, rad: 0.26, w: 4 },
   199→   199→  ];
   200→   200→
   201→   201→  const microaneurysms = scatterPoints(Math.min(cfg.maCount, 60), rng, 0.006, maZones);
   202→   202→  const hemorrhages = scatterPoints(Math.min(cfg.hemCount, 24), rng, 0.016, hemZones);
   203→   203→  const exudates = cfg.dmeRisk
   204→   204→    ? [...ringPoints(macula.cx, macula.cy, 0.11, Math.min(14, cfg.exCount), rng, 0.012),
   205→   205→       ...scatterPoints(Math.max(0, Math.min(cfg.exCount - 14, 20)), rng, 0.01, [{ cx: macula.cx, cy: macula.cy, rad: 0.3, w: 3 }, { cx: disc.cx, cy: disc.cy, rad: 0.24, w: 2 }])]
   206→   206→    : scatterPoints(Math.min(cfg.exCount, 22), rng, 0.011, [{ cx: disc.cx, cy: disc.cy, rad: 0.24, w: 3 }, { cx: macula.cx, cy: macula.cy, rad: 0.34, w: 2 }]);
   207→   207→
   208→   208→  return { microaneurysms, hemorrhages, exudates };
   209→   209→}

   211→   211→export function buildGradcam(patientId: string, severity: number, dmeRisk: boolean): EvidenceResult["gradcam"] {
   212→   212→  const rng = seededRandom(`gradcam:${patientId}`);
   213→   213→  if (severity === 0) {
   214→   214→    // healthy model looks at disc/vessels faintly
   215→   215→    return { cx: 0.3, cy: 0.38, rx: 0.14, ry: 0.13, intensity: 0.32 };
   216→   216→  }
   217→   217→  if (dmeRisk) {
   218→   218→    return { cx: 0.62, cy: 0.55, rx: 0.16, ry: 0.15, intensity: 0.95 };
   219→   219→  }
   220→   220→  const cx = 0.34 + rng() * 0.3;
   221→   221→  const cy = 0.4 + rng() * 0.25;
   222→   222→  return { cx, cy, rx: 0.12 + severity * 0.025, ry: 0.11 + severity * 0.025, intensity: 0.55 + severity * 0.1 };
   223→   223→}

   380→   380→      timings_ms: base.timings_ms,
   381→   381→    };
   382→   382→    return withLesions;
   383→   383→  }
   384→   384→
   385→   385→  // Unknown patient: deterministic synthesis seeded by id (+ optional image bytes hash)
   386→   386→  const rng = seededRandom(`pipeline:${patientId}:${imageHint ?? ""}`);
   387→   387→  const quality = clamp(0.55 + rng() * 0.45, 0.55, 1);
   388→   388→  const severity = quality < 0.62 ? (rng() > 0.5 ? 1 : 2) : Math.floor(rng() * 5);
   389→   389→  const accept = quality >= 0.6;
   390→   390→

   403→   403→    };

   405→   405→  }
   406→   406→
   407→   407→  const cfgs: Record<number, { ma: number; hem: number; ex: number; vd: number; dme: boolean }> = {
   408→   408→    0: { ma: 0, hem: 0, ex: 0, vd: 8.9, dme: false },

   411→   411→    3: { ma: 84, hem: 27, ex: 19, vd: 12.7, dme: rng() > 0.4 },
   412→   412→    4: { ma: 131, hem: 52, ex: 34, vd: 14.6, dme: true },
   413→   413→  };
   414→   414→  const cfg = cfgs[severity];
   415→   415→  const conf = clamp(0.62 + rng() * 0.35, 0.62, 0.97);
   416→   416→  const consistency = clamp(0.72 + rng() * 0.26, 0.72, 0.98);
   417→   417→  const overlap = clamp(0.62 + rng() * 0.38, 0.62, 1);
   418→   418→  const centroid = clamp(0.15 + rng() * 0.9, 0.15, 1.05);
   419→   419→
   420→   420→  const probs: Record<string, number> = {};
   421→   421→  const raw = PROB_LABELS.map((_, i) => (i === severity ? conf : rng() * (severity > i ? 0.25 : 0.22)));
   422→   422→  const sum = raw.reduce((a, b) => a + b, 0);
   423→   423→  PROB_LABELS.forEach((l, i) => (probs[l] = Number((raw[i] / sum).toFixed(3))));
   424→   424→  probs[PROB_LABELS[severity]] = Number(conf.toFixed(3));
   425→   425→

   428→   428→  const trustLevel: TrustLevel = trustScore >= TRUST_THRESHOLDS.HIGH ? "HIGH" : trustScore >= TRUST_THRESHOLDS.MODERATE_LOW ? "MODERATE" : "LOW";

   430→   430→    severity >= 2 && dme ? "URGENT" : trustLevel === "HIGH" ? (severity >= 2 ? "NEEDS_REVIEW" : "AUTO_CLEARED") : trustLevel === "MODERATE" ? "NEEDS_REVIEW" : "NEEDS_REVIEW";

   432→   432→    status === "URGENT"
   433→   433→      ? "URGENT — DME suspected: immediate referral + doctor review"
   434→   434→      : trustLevel === "HIGH"
   435→   435→        ? "TRUSTED — auto screening recommendation"
   436→   436→        : "REVIEW — queued for ophthalmologist sign-off";
   437→   437→
   438→   438→  return {
   439→   439→    patient_id: patientId,
   440→   440→    created_at: new Date().toISOString(),
   441→   441→    gate: { quality_score: Number(quality.toFixed(2)), accepted: true, enhanced: quality < 0.75, message: quality < 0.75 ? "Usable after CLAHE enhancement" : "Sharp fundus image — PASS" },
   442→   442→    evidence: {

   444→   444→      dme_risk: dme,

   446→   446→      lesions: buildLesions(patientId, { maCount: cfg.ma, hemCount: cfg.hem, exCount: cfg.ex, dmeRisk: dme, severity }),
   447→   447→      gradcam: buildGradcam(patientId, severity, dme),
   448→   448→    },
   449→   449→    classification: { predicted_class: ICDR_CLASSES[Math.max(0, severity)].short, class_level: severity, confidence: Number(conf.toFixed(3)), probabilities: probs },
   450→   450→    explainability: { consistency: Number(consistency.toFixed(3)), verdict: consistency >= TRUST_THRESHOLDS.HIGH ? "HIGH" : "MODERATE", centroid_distance_dd: Number(centroid.toFixed(2)), region_overlap: Number(overlap.toFixed(2)) },
   451→   451→    trust: { trust_score: Number(trustScore.toFixed(3)), trust_level: trustLevel, route },
   452→   452→    status,
   453→   453→    report_url: `/api/patients/${patientId}`,
   454→   454→    timings_ms: timings(700 + Math.round(rng() * 150), 1250 + Math.round(rng() * 300), 1000 + Math.round(rng() * 250), 1350 + Math.round(rng() * 300)),
   455→   455→  };

   504→   504→// ────────────────────────────────────────────────────────────
   505→   505→// Capacity planner math (M/M/c queueing — mirrors module5_capacity_planner.py)
   506→   506→// ────────────────────────────────────────────────────────────
   507→   507→
   508→   508→export interface CapacityInput {
   509→   509→  cams: number; // cameras (screening stations) 1-10
   510→   510→  revw: number; // reviewers (ophthalmologists/graders) 1-6
   511→   511→  arr: number; // arrivals per hour 10-60
   512→   512→}
   513→   513→
   514→   514→export interface CapacityOutput {
   515→   515→  patientsPerDay: number;
   516→   516→  patientsPerYear: number;
   517→   517→  meanWaitMin: number;
   518→   518→  utilizationPct: number;
   519→   519→  reviewUtilizationPct: number;
   520→   520→  reviewWaitMin: number;
   521→   521→  bottleneck: "CAPTURE" | "REVIEW" | "BALANCED";
   522→   522→  queueLength: number;
   523→   523→  serviceMinPerPatient: number;
   524→   524→}
   525→   525→
   526→   526→function erlangC(lambda: number, mu: number, c: number): { wq: number; rho: number } {
   527→   527→  // lambda: arrival rate /hr, mu: service rate per server /hr, c: servers
   528→   528→  const a = lambda / mu; // offered load
   529→   529→  if (a >= c) return { wq: Infinity, rho: 1 };
   530→   530→  const rho = a / c;
   531→   531→  // Erlang C formula
   532→   532→  let sum = 0;
   533→   533→  for (let n = 0; n < c; n++) sum += Math.pow(a, n) / factorial(n);
   534→   534→  const last = Math.pow(a, c) / (factorial(c) * (1 - rho));
   535→   535→  const cProb = last / (sum + last);
   536→   536→  const wq = cProb / (c * mu - lambda); // hours
   537→   537→  return { wq: wq * 60, rho }; // minutes
   538→   538→}
   539→   539→
   540→   540→function factorial(n: number): number {
   541→   541→  let f = 1;
   542→   542→  for (let i = 2; i <= n; i++) f *= i;
   543→   543→  return f;
   544→   544→}

   546→   546→export const CAPACITY_PARAMS = {
   547→   547→  captureMinPerPatient: 4, // minutes per capture at a camera station
   548→   548→  reviewMinPerCase: 2.5, // minutes of reviewer time per case
   549→   549→  hoursPerDay: 8,
   550→   550→  workingDays: 250,
   551→   551→};
   552→   552→
   553→   553→export function computeCapacity(input: CapacityInput): CapacityOutput {
   554→   554→  const { cams, revw, arr } = input;
   555→   555→  const { captureMinPerPatient, reviewMinPerCase, hoursPerDay, workingDays } = CAPACITY_PARAMS;
   556→   556→
   557→   557→  const lambda = arr; // per hour
   558→   558→  const captureMu = 60 / captureMinPerPatient; // per station per hour
   559→   559→  const captureCap = cams * captureMu;
   560→   560→
   561→   561→  // capture stage: effectively M/M/c with c = cams
   562→   562→  const cap = erlangC(lambda, captureMu, cams);
   563→   563→  const captureUtil = clamp(lambda / captureCap, 0, 1);
   564→   564→
   565→   565→  // review stage: high-trust cases auto-cleared (~65%), others need human review
   566→   566→  const reviewLambda = lambda * 0.35;
   567→   567→  const reviewMu = 60 / reviewMinPerCase;
   568→   568→  const rev = erlangC(reviewLambda, reviewMu, revw);
   569→   569→  const reviewUtil = clamp(reviewLambda / (revw * reviewMu), 0, 1);
   570→   570→
   571→   571→  const throughput = Math.min(lambda, captureCap);
   572→   572→  const patientsPerDay = Math.round(throughput * hoursPerDay);
   573→   573→  const patientsPerYear = patientsPerDay * workingDays;
   574→   574→
   575→   575→  const waitMin = isFinite(cap.wq) ? cap.wq : 999;
   576→   576→  const reviewWaitMin = isFinite(rev.wq) ? rev.wq : 999;
   577→   577→  const meanWaitMin = waitMin + (throughput > 0 ? reviewWaitMin * 0.35 : 0);
   578→   578→
   579→   579→  const utilizationPct = Math.round(captureUtil * 100);
   580→   580→  const bottleneck: CapacityOutput["bottleneck"] =
   581→   581→    captureUtil > reviewUtil + 0.08 ? "CAPTURE" : reviewUtil > captureUtil + 0.08 ? "REVIEW" : "BALANCED";
   582→   582→
   583→   583→  // Little's law for queue visualization
   584→   584→  const queueLength = isFinite(cap.wq) ? (lambda * (cap.wq / 60)) : 30;
   585→   585→
   586→   586→  return {
   587→   587→    patientsPerDay,
   588→   588→    patientsPerYear,
   589→   589→    meanWaitMin: Math.round(meanWaitMin * 10) / 10,
   590→   590→    utilizationPct,
   591→   591→    reviewUtilizationPct: Math.round(reviewUtil * 100),
   592→   592→    reviewWaitMin: Math.round(reviewWaitMin * 10) / 10,
   593→   593→    bottleneck,
   594→   594→    queueLength: Math.round(queueLength * 10) / 10,
   595→   595→    serviceMinPerPatient: captureMinPerPatient,
   596→   596→  };
   597→   597→}

   605→   605→/** District scaling: patients/year across N district deployments */
   606→   606→export function districtScaling(cams: number, revw: number, arr: number) {
   607→   607→  const out: Array<{ districts: number; patientsPerYear: number }> = [];
   608→   608→  for (let d = 1; d <= 20; d++) {
   609→   609→    const per = computeCapacity({ cams, revw, arr }).patientsPerYear;
   610→   610→    out.push({ districts: d, patientsPerYear: per * d });
   611→   611→  }
   612→   612→  return out;
   613→   613→}
