   108→function fmtDate(iso: string): string {
   109→  const d = new Date(iso);
   110→  if (Number.isNaN(d.getTime())) return "—";
   111→  return d.toLocaleString("en-US", { month: "short", day: "numeric", hour: "2-digit", minute: "2-digit", hour12: false });
   112→}
   113→
   114→function icdrColor(level: number): string | undefined {
   115→  return ICDR_CLASSES.find((c) => c.level === level)?.color;

   118→function rowGradeColor(row: PatientRow): string | undefined {
   119→  if (row.status === "REJECTED" || row.class_level < 0) return undefined;
   120→  return icdrColor(row.class_level);

   255→export default function DashboardView() {
   256→  const { navigate } = useNav();
   257→
   258→  // full dataset (stats + tab counts)
   259→  const [allRows, setAllRows] = useState<PatientRow[] | null>(null);
   260→  const [allLoading, setAllLoading] = useState(true);
   261→  const [allError, setAllError] = useState<string | null>(null);
   262→
   263→  // current filter queue
   264→  const [filter, setFilter] = useState<FilterKey>("all");
   265→  const [rows, setRows] = useState<PatientRow[] | null>(null);
   266→  const [rowsLoading, setRowsLoading] = useState(true);
   267→  const [rowsError, setRowsError] = useState<string | null>(null);
   268→
   269→  // report modal
   270→  const [openId, setOpenId] = useState<string | null>(null);
   271→  const [detail, setDetail] = useState<PatientDetail | null>(null);
   272→  const [detailLoading, setDetailLoading] = useState(false);
   273→  const [detailError, setDetailError] = useState<string | null>(null);
   274→  const [detailNonce, setDetailNonce] = useState(0);
   275→
   276→  // sign-off audit trail: patientId → who/when (persisted server-side via PATCH)
   277→  const signedOffRef = useRef<Record<string, boolean>>({});
   278→  const [signedOff, setSignedOff] = useState<Record<string, { by: string; at: string; previousStatus: CaseStatus }>>({});
   279→  const [signingOff, setSigningOff] = useState<string | null>(null);
   280→  const [reopenTarget, setReopenTarget] = useState<string | null>(null);
   281→  const quietRefreshRef = useRef(false);
   282→
   283→  const applyOverrides = useCallback(
   284→    (list: PatientRow[]): PatientRow[] =>
   285→      list.map((r) => (signedOffRef.current[r.patient_id] ? { ...r, status: "AUTO_CLEARED" as CaseStatus } : r)),
   286→    []
   287→  );
   288→
   289→  const loadAll = useCallback(
   290→    async (silent = false) => {
   291→      if (!silent) setAllLoading(true);
   292→      setAllError(null);
   293→      try {
   294→        const res = await fetch("/api/patients");
   295→        if (!res.ok) throw new Error(`API returned ${res.status}`);
   296→        const data = (await res.json()) as { patients: PatientRow[] };
   297→        setAllRows(applyOverrides(data.patients ?? []));
   298→      } catch (e) {
   299→        quietRefreshRef.current = false;
   300→        setAllError(e instanceof Error ? e.message : "Failed to load cases");
   301→      } finally {
   302→        setAllLoading(false);
   303→      }
   304→    },
   305→    [applyOverrides]
   306→  );
   307→
   308→  useEffect(() => {
   309→    void loadAll();
   310→  }, [loadAll]);
   311→
   312→  // queue rows follow the active filter; "all" reuses the already-loaded list
   313→  useEffect(() => {
   314→    if (filter === "all" && allRows) {
   315→      setRows(allRows);
   316→      setRowsLoading(false);
   317→      setRowsError(null);
   318→      return;
   319→    }
   320→    const quiet = quietRefreshRef.current;
   321→    quietRefreshRef.current = false;
   322→    let cancelled = false;
   323→    if (!quiet) setRowsLoading(true);
   324→    setRowsError(null);
   325→    fetch(`/api/patients?filter=${filter}`)
   326→      .then(async (res) => {
   327→        if (!res.ok) throw new Error(`API returned ${res.status}`);
   328→        return (await res.json()) as { patients: PatientRow[] };
   329→      })
   330→      .then((d) => {
   331→        if (!cancelled) setRows(applyOverrides(d.patients ?? []));
   332→      })
   333→      .catch((e) => {
   334→        if (!cancelled) setRowsError(e instanceof Error ? e.message : "Failed to load queue");
   335→      })
   336→      .finally(() => {
   337→        if (!cancelled && !quiet) setRowsLoading(false);
   338→        if (quiet) setRowsLoading(false);
   339→      });
   340→    return () => {
   341→      cancelled = true;
   342→    };
   343→  }, [filter, allRows, applyOverrides]);
   344→
   345→  // report detail fetch on modal open / retry
   346→  useEffect(() => {
   347→    if (!openId) return;
   348→    let cancelled = false;
   349→    setDetailLoading(true);
   350→    setDetailError(null);
   351→    setDetail(null);
   352→    fetch(`/api/patients/${encodeURIComponent(openId)}`)
   353→      .then(async (res) => {
   354→        if (!res.ok) throw new Error(`API returned ${res.status}`);
   355→        return (await res.json()) as PatientDetail;
   356→      })
   357→      .then((d) => {
   358→        if (cancelled) return;
   359→        if (signedOffRef.current[openId] && d.details) {
   360→          setDetail({ ...d, status: "AUTO_CLEARED", details: { ...d.details, status: "AUTO_CLEARED" } });
   361→        } else {
   362→          setDetail(d);
   363→        }
   364→      })
   365→      .catch((e) => {
   366→        if (!cancelled) setDetailError(e instanceof Error ? e.message : "Failed to load report");
   367→      })
   368→      .finally(() => {
   369→        if (!cancelled) setDetailLoading(false);
   370→      });
   371→    return () => {
   372→      cancelled = true;
   373→    };
   374→  }, [openId, detailNonce]);
   375→
   376→  const stats = useMemo(() => {
   377→    const list = allRows ?? [];
   378→    const today = new Date().toDateString();
   379→    const screenedToday = list.filter((r) => new Date(r.created_at).toDateString() === today).length;
   380→    const referable = list.filter((r) => r.class_level >= 2 && r.status !== "REJECTED").length;
   381→    const urgentCount = list.filter((r) => r.status === "URGENT").length;
   382→    const queue = list.filter((r) => r.status === "NEEDS_REVIEW" || r.status === "URGENT").length;
   383→    const avgMs = list.length ? list.reduce((a, r) => a + r.processing_ms, 0) / list.length : 0;
   384→    return { screenedToday, referable, urgentCount, queue, avgSec: avgMs / 1000 };
   385→  }, [allRows]);
   386→
   387→  const counts = useMemo(() => {
   388→    const list = allRows ?? [];
   389→    const c: Record<FilterKey, number> = { all: list.length, auto_cleared: 0, needs_review: 0, urgent: 0, rejected: 0 };
   390→    for (const r of list) {
   391→      if (r.status === "AUTO_CLEARED") c.auto_cleared++;
   392→      else if (r.status === "NEEDS_REVIEW") c.needs_review++;
   393→      else if (r.status === "URGENT") c.urgent++;
   394→      else if (r.status === "REJECTED") c.rejected++;
   395→    }
   396→    return c;
   397→  }, [allRows]);
   398→
   399→  const openReport = useCallback((patientId: string) => {
   400→    setOpenId(patientId);
   401→    setDetailNonce((n) => n + 1);
   402→  }, []);
   403→
   404→  async function handleSignOff(patientId: string) {
   405→    if (signingOff) return;
   406→    setSigningOff(patientId);
   407→    try {
   408→      const res = await fetch(`/api/patients/${encodeURIComponent(patientId)}/status`, {
   409→        method: "PATCH",
   410→        headers: { "Content-Type": "application/json" },
   411→        body: JSON.stringify({ status: "AUTO_CLEARED", reviewed_by: "Dr. Review (dashboard demo)" }),
   412→      });
   413→      if (!res.ok) {
   414→        const err = (await res.json().catch(() => ({}))) as { error?: string };
   415→        throw new Error(err.error ?? `API returned ${res.status}`);
   416→      }
   417→      const data = (await res.json()) as { reviewed_by: string; reviewed_at: string };
   418→      const at = data.reviewed_at ? new Date(data.reviewed_at).toLocaleString() : new Date().toLocaleString();
   419→      toast.success(`Signed off by ${data.reviewed_by} — saved to the register`);
   420→      signedOffRef.current = { ...signedOffRef.current, [patientId]: true };
   421→      // remember the pre-sign-off status so Undo can restore the right queue lane
   422→      const prevRow = allRows?.find((r) => r.patient_id === patientId) ?? rows?.find((r) => r.patient_id === patientId);
   423→      const previousStatus: CaseStatus =
   424→        prevRow && (prevRow.status === "NEEDS_REVIEW" || prevRow.status === "URGENT")
   425→          ? prevRow.status
   426→          : detailResult && (detailResult.status === "NEEDS_REVIEW" || detailResult.status === "URGENT")
   427→            ? detailResult.status
   428→            : "NEEDS_REVIEW";
   429→      setSignedOff((s) => ({ ...s, [patientId]: { by: data.reviewed_by, at, previousStatus } }));
   430→      const flip = (r: PatientRow): PatientRow =>
   431→        r.patient_id === patientId
   432→          ? { ...r, status: "AUTO_CLEARED", reviewed_by: data.reviewed_by, reviewed_at: data.reviewed_at }
   433→          : r;
   434→      setAllRows((prev) => (prev ? applyOverrides(prev.map(flip)) : prev));
   435→      setRows((prev) => (prev ? applyOverrides(prev.map(flip)) : prev));
   436→      setDetail((prev) =>
   437→        prev && prev.details
   438→          ? { ...prev, status: "AUTO_CLEARED", details: { ...prev.details, status: "AUTO_CLEARED" } }
   439→          : prev
   440→      );
   441→      // quiet refresh of stats + counts (list already flipped optimistically)
   442→      quietRefreshRef.current = true;
   443→      void loadAll(true);
   444→    } catch (e) {
   445→      toast.error(e instanceof Error ? e.message : "Sign-off failed — try again");
   446→    } finally {
   447→      setSigningOff(null);
   448→    }
   449→  }
   450→
   451→  async function handleReopen(patientId: string) {
   452→    if (signingOff) return;
   453→    const record = signedOff[patientId];
   454→    // post-reload undo can't know the original lane — infer from routing rules
   455→    // (URGENT = LOW-trust or DME-risk cases jump the queue; everything else is MODERATE review)
   456→    const previousStatus: CaseStatus =
   457→      record?.previousStatus ??
   458→      (detailResult && (detailResult.trust.trust_level === "LOW" || detailResult.evidence.dme_risk) ? "URGENT" : "NEEDS_REVIEW");
   459→    setSigningOff(patientId);
   460→    try {
   461→      const res = await fetch(`/api/patients/${encodeURIComponent(patientId)}/status`, {
   462→        method: "PATCH",
   463→        headers: { "Content-Type": "application/json" },
   464→        body: JSON.stringify({
   465→          status: previousStatus,
   466→          reviewed_by: "Dr. Review (dashboard demo)",
   467→          note: `Sign-off reopened by Dr. Review — case returned to the ${previousStatus === "URGENT" ? "urgent" : "review"} queue`,
   468→        }),
   469→      });
   470→      if (!res.ok) {
   471→        const err = (await res.json().catch(() => ({}))) as { error?: string };
   472→        throw new Error(err.error ?? `API returned ${res.status}`);
   473→      }
   474→      toast.success(`Case reopened — returned to the ${previousStatus === "URGENT" ? "urgent" : "review"} queue`);
   475→      signedOffRef.current = Object.fromEntries(
   476→        Object.entries(signedOffRef.current).filter(([k]) => k !== patientId)
   477→      );
   478→      setSignedOff((s) => Object.fromEntries(Object.entries(s).filter(([k]) => k !== patientId)));
   479→      const flip = (r: PatientRow): PatientRow =>
   480→        r.patient_id === patientId
   481→          ? { ...r, status: previousStatus, reviewed_by: null, reviewed_at: null }
   482→          : r;
   483→      setAllRows((prev) => (prev ? prev.map(flip) : prev));
   484→      setRows((prev) => (prev ? prev.map(flip) : prev));
   485→      setDetail((prev) =>
   486→        prev && prev.details
   487→          ? { ...prev, status: previousStatus, details: { ...prev.details, status: previousStatus } }
   488→          : prev
   489→      );
   490→      // quiet refresh of stats + counts (list already flipped optimistically)
   491→      quietRefreshRef.current = true;
   492→      void loadAll(true);
   493→    } catch (e) {
   494→      toast.error(e instanceof Error ? e.message : "Reopen failed — try again");
   495→    } finally {
   496→      setSigningOff(null);
   497→      setReopenTarget(null);
   498→    }
   499→  }
   500→
   501→  const detailResult = detail?.details ?? null;
   502→  const isRejected = detailResult ? detailResult.status === "REJECTED" || detailResult.gate.accepted === false : false;
   503→  const canSign =
   504→    !!detailResult && (detailResult.status === "NEEDS_REVIEW" || detailResult.status === "URGENT") && !signedOff[openId ?? ""];
   505→  const isSignedOffHere = Boolean(signedOff[openId ?? ""] || detail?.reviewed_by);
   506→  const reviewedAtText = (() => {
   507→    const iso = detail?.reviewed_at ?? null;
   508→    if (!iso) return signedOff[openId ?? ""]?.at ?? "";
