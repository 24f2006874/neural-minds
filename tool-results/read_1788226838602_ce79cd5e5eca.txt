    79→/**
    80→ * Parse the shareable config from the hash query — `#/capacity?c=3-2-25&cmp=1`.
    81→ * Values outside the knob ranges are ignored (falls back to defaults).
    82→ */
    83→function readCapacityHash(): { cams?: number; revw?: number; arr?: number; cmp?: boolean } {
    84→  if (typeof window === "undefined") return {};
    85→  const raw = window.location.hash.replace(/^#\/?/, "");
    86→  const q = raw.indexOf("?");
    87→  if (q === -1) return {};
    88→  const params = new URLSearchParams(raw.slice(q + 1));
    89→  const out: { cams?: number; revw?: number; arr?: number; cmp?: boolean } = {};
    90→  const c = params.get("c");
    91→  if (c) {
    92→    const parts = c.split("-").map(Number);
    93→    if (parts.length === 3 && parts.every((n) => Number.isFinite(n))) {
    94→      const [cams, revw, arr] = parts;
    95→      if (cams >= 1 && cams <= 10) out.cams = cams;
    96→      if (revw >= 1 && revw <= 6) out.revw = revw;
    97→      if (arr >= 10 && arr <= 60) out.arr = arr;
    98→    }
    99→  }
   100→  if (params.get("cmp") === "1") out.cmp = true;
   101→  return out;
   102→}

   104→/** Trust-color load scale: green ≤75 · amber 75–90 · red >90. */
   105→function loadTone(pct: number): Tone {
   106→  if (pct > 90) return { text: "text-[#F87171] text-glow-red", chip: "border-[#F87171]/40 text-[#F87171]" };
   107→  if (pct > 75) return { text: "text-[#FBBF24]", chip: "border-[#FBBF24]/40 text-[#FBBF24]" };
   108→  return { text: "text-[#34D399]", chip: "border-[#34D399]/40 text-[#34D399]" };
   109→}
   110→
   111→/** Wait scale: green ≤15 min · amber >30 · red >60 (or unbounded). */
   112→function waitTone(min: number): Tone {
   113→  if (min >= UNBOUNDED_WAIT || min > 60)
   114→    return { text: "text-[#F87171] text-glow-red", chip: "border-[#F87171]/40 text-[#F87171]" };
   115→  if (min > 30) return { text: "text-[#FBBF24]", chip: "border-[#FBBF24]/40 text-[#FBBF24]" };
   116→  if (min <= 15) return { text: "text-[#34D399]", chip: "border-[#34D399]/40 text-[#34D399]" };
   117→  return { text: "text-[#22D3EE]", chip: "border-[#22D3EE]/40 text-[#22D3EE]" };
   118→}

   120→/** Decorative flow dots — travel left → right, speed set by `duration`. */
   121→function FlowDots({
   122→  count,
   123→  duration,
   124→  color,
   125→  reduced,
   126→}: {
   127→  count: number;
   128→  duration: number;
   129→  color: string;
   130→  reduced: boolean;
   131→}) {
   132→  const dots = Array.from({ length: count }, (_, i) => i);
   133→  if (reduced) {
   134→    return (
   135→      <>
   136→        {dots.map((i) => (
   137→          <span
   138→            key={i}
   139→            className="absolute h-1.5 w-1.5 rounded-full"
   140→            style={{
   141→              top: "calc(50% - 3px)",
   142→              left: `${count > 1 ? 8 + (i * 84) / (count - 1) : 46}%`,
   143→              background: color,
   144→              opacity: 0.55,
   145→            }}
   146→          />
   147→        ))}
   148→      </>
   149→    );
   150→  }
   151→  return (
   152→    <>
   153→      {dots.map((i) => (
   154→        <motion.span
   155→          key={i}
   156→          className="absolute h-1.5 w-1.5 rounded-full"
   157→          style={{ top: "calc(50% - 3px)", background: color, boxShadow: `0 0 8px ${color}` }}
   158→          initial={{ left: "-3%" }}
   159→          animate={{ left: "103%" }}
   160→          transition={{ duration, repeat: Infinity, ease: "linear", delay: (i * duration) / count }}
   161→        />
   162→      ))}
   163→    </>
   164→  );
   165→}
