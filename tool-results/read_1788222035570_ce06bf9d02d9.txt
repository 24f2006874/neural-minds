    72→function clamp(v: number, lo: number, hi: number) {
    73→  return Math.min(hi, Math.max(lo, v));
    74→}
    75→
    76→/** Trust-color load scale: green ≤75 · amber 75–90 · red >90. */
    77→function loadTone(pct: number): Tone {
    78→  if (pct > 90) return { text: "text-[#F87171] text-glow-red", chip: "border-[#F87171]/40 text-[#F87171]" };
    79→  if (pct > 75) return { text: "text-[#FBBF24]", chip: "border-[#FBBF24]/40 text-[#FBBF24]" };
    80→  return { text: "text-[#34D399]", chip: "border-[#34D399]/40 text-[#34D399]" };
    81→}
    82→
    83→/** Wait scale: green ≤15 min · amber >30 · red >60 (or unbounded). */
    84→function waitTone(min: number): Tone {
    85→  if (min >= UNBOUNDED_WAIT || min > 60)
    86→    return { text: "text-[#F87171] text-glow-red", chip: "border-[#F87171]/40 text-[#F87171]" };
    87→  if (min > 30) return { text: "text-[#FBBF24]", chip: "border-[#FBBF24]/40 text-[#FBBF24]" };
    88→  if (min <= 15) return { text: "text-[#34D399]", chip: "border-[#34D399]/40 text-[#34D399]" };
    89→  return { text: "text-[#22D3EE]", chip: "border-[#22D3EE]/40 text-[#22D3EE]" };
    90→}

    92→/** Decorative flow dots — travel left → right, speed set by `duration`. */
    93→function FlowDots({
    94→  count,
    95→  duration,
    96→  color,
    97→  reduced,
    98→}: {
    99→  count: number;
   100→  duration: number;
   101→  color: string;
   102→  reduced: boolean;
   103→}) {
   104→  const dots = Array.from({ length: count }, (_, i) => i);
   105→  if (reduced) {
   106→    return (
   107→      <>
   108→        {dots.map((i) => (
   109→          <span
   110→            key={i}
   111→            className="absolute h-1.5 w-1.5 rounded-full"
   112→            style={{
   113→              top: "calc(50% - 3px)",
   114→              left: `${count > 1 ? 8 + (i * 84) / (count - 1) : 46}%`,
   115→              background: color,
   116→              opacity: 0.55,
   117→            }}
   118→          />
   119→        ))}
   120→      </>
   121→    );
   122→  }
   123→  return (
   124→    <>
   125→      {dots.map((i) => (
   126→        <motion.span
   127→          key={i}
   128→          className="absolute h-1.5 w-1.5 rounded-full"
   129→          style={{ top: "calc(50% - 3px)", background: color, boxShadow: `0 0 8px ${color}` }}
   130→          initial={{ left: "-3%" }}
   131→          animate={{ left: "103%" }}
   132→          transition={{ duration, repeat: Infinity, ease: "linear", delay: (i * duration) / count }}
   133→        />
   134→      ))}
   135→    </>
   136→  );
   137→}
