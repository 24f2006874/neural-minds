    97→function fmtDate(iso: string): string {
    98→  const d = new Date(iso);
    99→  if (Number.isNaN(d.getTime())) return "—";
   100→  return d.toLocaleString("en-US", { month: "short", day: "numeric", hour: "2-digit", minute: "2-digit", hour12: false });
   101→}
   102→
   103→function icdrColor(level: number): string | undefined {
   104→  return ICDR_CLASSES.find((c) => c.level === level)?.color;
   105→}
   106→
   107→function rowGradeColor(row: PatientRow): string | undefined {
   108→  if (row.status === "REJECTED" || row.class_level < 0) return undefined;
   109→  return icdrColor(row.class_level);
   110→}

   244→export default function DashboardView() {
   245→  const { navigate } = useNav();
   246→
   247→  // full dataset (stats + tab counts)
   248→  const [allRows, setAllRows] = useState<PatientRow[] | null>(null);
   249→  const [allLoading, setAllLoading] = useState(true);
   250→  const [allError, setAllError] = useState<string | null>(null);
   251→
   252→  // current filter queue
   253→  const [filter, setFilter] = useState<FilterKey>("all");
   254→  const [rows, setRows] = useState<PatientRow[] | null>(null);
   255→  const [rowsLoading, setRowsLoading] = useState(true);
   256→  const [rowsError, setRowsError] = useState<string | null>(null);
   257→
   258→  // report modal
   259→  const [openId, setOpenId] = useState<string | null>(null);
   260→  const [detail, setDetail] = useState<PatientDetail | null>(null);
   261→  const [detailLoading, setDetailLoading] = useState(false);
   262→  const [detailError, setDetailError] = useState<string | null>(null);
   263→  const [detailNonce, setDetailNonce] = useState(0);
   264→
   265→  // sign-off audit trail: patientId → who/when (persisted server-side via PATCH)
   266→  const signedOffRef = useRef<Record<string, boolean>>({});
   267→  const [signedOff, setSignedOff] = useState<Record<string, { by: string; at: string }>>({});
   268→  const [signingOff, setSigningOff] = useState<string | null>(null);
   269→  const quietRefreshRef = useRef(false);
   270→
   271→  const applyOverrides = useCallback(
   272→    (list: PatientRow[]): PatientRow[] =>
   273→      list.map((r) => (signedOffRef.current[r.patient_id] ? { ...r, status: "AUTO_CLEARED" as CaseStatus } : r)),
   274→    []
   275→  );
   276→
   277→  const loadAll = useCallback(
   278→    async (silent = false) => {
   279→      if (!silent) setAllLoading(true);
   280→      setAllError(null);
   281→      try {
   282→        const res = await fetch("/api/patients");
   283→        if (!res.ok) throw new Error(`API returned ${res.status}`);
   284→        const data = (await res.json()) as { patients: PatientRow[] };
   285→        setAllRows(applyOverrides(data.patients ?? []));
   286→      } catch (e) {
   287→        quietRefreshRef.current = false;
   288→        setAllError(e instanceof Error ? e.message : "Failed to load cases");
   289→      } finally {
   290→        setAllLoading(false);
   291→      }
   292→    },
   293→    [applyOverrides]
   294→  );
   295→
   296→  useEffect(() => {
   297→    void loadAll();
   298→  }, [loadAll]);
   299→
   300→  // queue rows follow the active filter; "all" reuses the already-loaded list
   301→  useEffect(() => {
   302→    if (filter === "all" && allRows) {
   303→      setRows(allRows);
   304→      setRowsLoading(false);
   305→      setRowsError(null);
   306→      return;
   307→    }
   308→    const quiet = quietRefreshRef.current;
   309→    quietRefreshRef.current = false;
   310→    let cancelled = false;
   311→    if (!quiet) setRowsLoading(true);
   312→    setRowsError(null);
   313→    fetch(`/api/patients?filter=${filter}`)
   314→      .then(async (res) => {
   315→        if (!res.ok) throw new Error(`API returned ${res.status}`);
   316→        return (await res.json()) as { patients: PatientRow[] };
   317→      })
   318→      .then((d) => {
   319→        if (!cancelled) setRows(applyOverrides(d.patients ?? []));
   320→      })
   321→      .catch((e) => {
   322→        if (!cancelled) setRowsError(e instanceof Error ? e.message : "Failed to load queue");
   323→      })
   324→      .finally(() => {
   325→        if (!cancelled && !quiet) setRowsLoading(false);
   326→        if (quiet) setRowsLoading(false);
   327→      });
   328→    return () => {
   329→      cancelled = true;
   330→    };
   331→  }, [filter, allRows, applyOverrides]);
   332→
   333→  // report detail fetch on modal open / retry
   334→  useEffect(() => {
   335→    if (!openId) return;
   336→    let cancelled = false;
   337→    setDetailLoading(true);
   338→    setDetailError(null);
   339→    setDetail(null);
   340→    fetch(`/api/patients/${encodeURIComponent(openId)}`)
   341→      .then(async (res) => {
   342→        if (!res.ok) throw new Error(`API returned ${res.status}`);
   343→        return (await res.json()) as PatientDetail;
   344→      })
   345→      .then((d) => {
   346→        if (cancelled) return;
   347→        if (signedOffRef.current[openId] && d.details) {
   348→          setDetail({ ...d, status: "AUTO_CLEARED", details: { ...d.details, status: "AUTO_CLEARED" } });
   349→        } else {
   350→          setDetail(d);
   351→        }
   352→      })
   353→      .catch((e) => {
   354→        if (!cancelled) setDetailError(e instanceof Error ? e.message : "Failed to load report");
   355→      })
   356→      .finally(() => {
   357→        if (!cancelled) setDetailLoading(false);
   358→      });
   359→    return () => {
   360→      cancelled = true;
   361→    };
   362→  }, [openId, detailNonce]);
   363→
   364→  const stats = useMemo(() => {
   365→    const list = allRows ?? [];
   366→    const today = new Date().toDateString();
   367→    const screenedToday = list.filter((r) => new Date(r.created_at).toDateString() === today).length;
   368→    const referable = list.filter((r) => r.class_level >= 2 && r.status !== "REJECTED").length;
   369→    const urgentCount = list.filter((r) => r.status === "URGENT").length;
   370→    const queue = list.filter((r) => r.status === "NEEDS_REVIEW" || r.status === "URGENT").length;
   371→    const avgMs = list.length ? list.reduce((a, r) => a + r.processing_ms, 0) / list.length : 0;
   372→    return { screenedToday, referable, urgentCount, queue, avgSec: avgMs / 1000 };
   373→  }, [allRows]);
   374→
   375→  const counts = useMemo(() => {
   376→    const list = allRows ?? [];
   377→    const c: Record<FilterKey, number> = { all: list.length, auto_cleared: 0, needs_review: 0, urgent: 0, rejected: 0 };
   378→    for (const r of list) {
   379→      if (r.status === "AUTO_CLEARED") c.auto_cleared++;
   380→      else if (r.status === "NEEDS_REVIEW") c.needs_review++;
   381→      else if (r.status === "URGENT") c.urgent++;
   382→      else if (r.status === "REJECTED") c.rejected++;
   383→    }
   384→    return c;
   385→  }, [allRows]);
   386→
   387→  const openReport = useCallback((patientId: string) => {
   388→    setOpenId(patientId);
   389→    setDetailNonce((n) => n + 1);
   390→  }, []);
   391→
   392→  async function handleSignOff(patientId: string) {
   393→    if (signingOff) return;
   394→    setSigningOff(patientId);
   395→    try {
   396→      const res = await fetch(`/api/patients/${encodeURIComponent(patientId)}/status`, {
   397→        method: "PATCH",
   398→        headers: { "Content-Type": "application/json" },
   399→        body: JSON.stringify({ status: "AUTO_CLEARED", reviewed_by: "Dr. Review (dashboard demo)" }),
   400→      });
   401→      if (!res.ok) {
   402→        const err = (await res.json().catch(() => ({}))) as { error?: string };
   403→        throw new Error(err.error ?? `API returned ${res.status}`);
   404→      }
   405→      const data = (await res.json()) as { reviewed_by: string; reviewed_at: string };
   406→      const at = data.reviewed_at ? new Date(data.reviewed_at).toLocaleString() : new Date().toLocaleString();
   407→      toast.success(`Signed off by ${data.reviewed_by} — saved to the register`);
   408→      signedOffRef.current = { ...signedOffRef.current, [patientId]: true };
   409→      setSignedOff((s) => ({ ...s, [patientId]: { by: data.reviewed_by, at } }));
   410→      const flip = (r: PatientRow): PatientRow =>
   411→        r.patient_id === patientId
   412→          ? { ...r, status: "AUTO_CLEARED", reviewed_by: data.reviewed_by, reviewed_at: data.reviewed_at }
   413→          : r;
   414→      setAllRows((prev) => (prev ? applyOverrides(prev.map(flip)) : prev));
   415→      setRows((prev) => (prev ? applyOverrides(prev.map(flip)) : prev));
   416→      setDetail((prev) =>
   417→        prev && prev.details
   418→          ? { ...prev, status: "AUTO_CLEARED", details: { ...prev.details, status: "AUTO_CLEARED" } }
   419→          : prev
   420→      );
   421→      // quiet refresh of stats + counts (list already flipped optimistically)
   422→      quietRefreshRef.current = true;
   423→      void loadAll(true);
   424→    } catch (e) {
   425→      toast.error(e instanceof Error ? e.message : "Sign-off failed — try again");
   426→    } finally {
   427→      setSigningOff(null);
   428→    }
   429→  }
   430→
   431→  const detailResult = detail?.details ?? null;
   432→  const isRejected = detailResult ? detailResult.status === "REJECTED" || detailResult.gate.accepted === false : false;
   433→  const canSign =
   434→    !!detailResult && (detailResult.status === "NEEDS_REVIEW" || detailResult.status === "URGENT") && !signedOff[openId ?? ""];
   435→  const reviewedAtText = (() => {
   436→    const iso = detail?.reviewed_at ?? null;
   437→    if (!iso) return signedOff[openId ?? ""]?.at ?? "";
   438→    try {
   439→      return new Date(iso).toLocaleString();
   440→    } catch {
   441→      return "";
   442→    }
   443→  })();
   444→  const reviewNoteText = detail?.reviewed_by
   445→    ? `Audit trail: signed off by ${detail.reviewed_by}${reviewedAtText ? ` at ${reviewedAtText}` : ""} — persisted in the screening register`
   446→    : "Audit trail: signed off — persisted in the screening register";
   447→
   448→  return (
   449→    <section className="mx-auto max-w-7xl px-4 py-20 sm:px-6">
   450→      <SectionHeading
   451→        eyebrow="HUMAN-IN-THE-LOOP"
   452→        title={
   453→          <>
   454→            Doctor Dashboard — <span className="text-glow-cyan">Review Queue</span>
   455→          </>
   456→        }
   457→        sub="HIGH-trust cases auto-clear. MODERATE cases wait for your sign-off. Urgent and DME cases jump the queue."
   458→      />
   459→      <Reveal className="-mt-4 mb-10 flex justify-center">
   460→        <span className="chip border-[#FBBF24]/40 bg-[#2A2210]/60 text-[#FBBF24]">
   461→          <TriangleAlert className="h-3 w-3" aria-hidden="true" />
   462→          Demo data — simulated screening records
   463→        </span>
   464→      </Reveal>
   465→
   466→      {/* 1 ── Stats cards */}
   467→      <Reveal delay={0.05}>
   468→        {allError ? (
   469→          <GlassCard className="border-[#F87171]/30 p-4 sm:p-6">
   470→            <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   471→              <div className="flex items-center gap-3">
   472→                <TriangleAlert className="h-5 w-5 shrink-0 text-[#F87171]" aria-hidden="true" />
   473→                <div>
   474→                  <p className="text-sm font-semibold text-foreground">Couldn&apos;t load dashboard stats</p>
   475→                  <p className="text-xs text-muted-foreground">{allError}</p>
   476→                </div>
   477→              </div>
   478→              <Button
   479→                size="sm"
   480→                variant="outline"
   481→                className="min-h-11 shrink-0 border-[#F87171]/40 text-[#F87171] hover:bg-[#F87171]/10 hover:text-[#F87171]"
   482→                onClick={() => void loadAll()}
   483→              >
   484→                <RefreshCw className="h-4 w-4" aria-hidden="true" />
   485→                Retry
   486→              </Button>
   487→            </div>
   488→          </GlassCard>
   489→        ) : (
   490→          <div className="grid grid-cols-2 gap-4 xl:grid-cols-4">
   491→            <StatCard
   492→              icon={ScanEye}
   493→              label="Screened today"
   494→              sub="cases through the DRISHTI pipeline today"
   495→              value={stats.screenedToday}
   496→              valueClass="text-[#22D3EE]"
   497→              iconClass="border-[#22D3EE]/30 bg-[#22D3EE]/10 text-[#22D3EE]"
   498→              loading={allLoading}
   499→            />
   500→            <StatCard
   501→              icon={Crosshair}
   502→              label="Referable caught"
   503→              sub="grade ≥ Moderate NPDR · refer within 3-6 months"
   504→              value={stats.referable}
   505→              valueClass="text-[#FBBF24]"
   506→              iconClass="border-[#FBBF24]/30 bg-[#FBBF24]/10 text-[#FBBF24]"
   507→              loading={allLoading}
   508→            />
   509→            <StatCard
   510→              icon={ClipboardList}
   511→              label="Review queue"
   512→              sub="MODERATE + URGENT awaiting doctor sign-off"
   513→              value={stats.queue}
